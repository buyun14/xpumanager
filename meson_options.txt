# Meson options for XPUM project

option('dev', type : 'boolean', value : false,
       description : 'Development mode: disable warnings as errors')

option('shared_library', type : 'boolean', value : false,
       description : 'Build and install the shared library (libxpum.so/.dll); when false, xpu-smi links libxpum statically')

# Dependency options
option('use_system_levelzero', type : 'boolean', value : false,
       description : '(Not preferred) Use system Level Zero installation instead of automatic management')

option('use_system_igsc', type : 'boolean', value : false,
       description : '(Not preferred) Use system IGSC installation instead of automatic management')

# ---- Not quite supported but in the meson files
option('static_library', type : 'boolean', value : false,
       description : 'Install the static library (libxpum.a/.lib); enabled automatically when shared_library=false is insufficient')

option('conan_deps_build_type', type: 'string', value: 'Release',
       description: 'Build type used for Conan dependencies')

option('conan_deps_runtime', type: 'string', value: 'Release',
       description: 'Runtime type used for Conan dependencies')

option('with_tests', type: 'boolean', value: false,
       description: 'Build unit tests (requires boost-ext-ut via Conan)')

option('extensions', type: 'boolean', value: false,
       description: 'Include extensions-only features (dev branch only)')
option('level_zero_sysman_compat', type : 'boolean', value : false,
       description : 'Provide stubs for SYSMAN APIs missing from older Level Zero loaders (zesDevicePciLinkSpeedUpdateExt, zesRasGetStateExp2, zesRasGetSupportedCategoriesExp). Enable when the installed loader cannot initialise the installed GPU driver, since a newer loader would fail zeInit() with ZE_RESULT_ERROR_UNSUPPORTED_VERSION. Leave disabled when the loader already exports these symbols.')
